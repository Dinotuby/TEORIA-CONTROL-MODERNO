% CONFIGURA TU PUERTO Y NOMBRE DE ARCHIVO
puerto = "COM8";     % Ajusta según tu caso
baudrate = 115200;
archivoCSV = "datos_pid.csv";

% ABRIR PUERTO SERIAL
s = serialport(puerto, baudrate);
configureTerminator(s, "LF");

disp("Esperando al Arduino...");
pause(3);  % Esperar que se reinicie

flush(s);  % Limpiar buffers

% ENVÍA COMANDO PARA INICIAR
disp("Enviando 'S' para comenzar...");
writeline(s, "S");

% INICIALIZAR VARIABLES
datos = [];
tiempos = [];
rpms = [];
pwms = [];

% CONFIGURAR GRAFICO EN TIEMPO REAL
figure;
yyaxis left;
h1 = plot(nan, nan, 'b-', 'LineWidth', 2); % RPM - Azul
ylabel('RPM');

yyaxis right;
h2 = plot(nan, nan, 'r--', 'LineWidth', 2); % PWM - Rojo
ylabel('PWM (%)');

xlabel('Tiempo [s]');
title('Lectura en tiempo real: RPM y PWM');
grid on;
xlim([0 10]);
ylim([0 255]);  % Límite máximo PWM
hold on;

% LEER DATOS
disp("Recibiendo y graficando...");
while true
    if s.NumBytesAvailable > 0
        linea = readline(s);
        if contains(linea, "FIN")
            break;
        end

        partes = split(strtrim(linea), ',');
        if numel(partes) == 3
            t = str2double(partes{1});
            rpm = str2double(partes{2});
            pwm = str2double(partes{3});
            if ~isnan(t) && ~isnan(rpm) && ~isnan(pwm)
                datos(end+1, :) = [t, rpm, pwm]; %#ok<AGROW>
                tiempos(end+1) = t;              %#ok<AGROW>
                rpms(end+1) = rpm;               %#ok<AGROW>
                pwms(end+1) = pwm;               %#ok<AGROW>

                % ACTUALIZAR GRAFICO
                yyaxis left;
                set(h1, 'XData', tiempos, 'YData', rpms);

                yyaxis right;
                set(h2, 'XData', tiempos, 'YData', pwms);

                drawnow limitrate;
            end
        end
    end
end

% CERRAR PUERTO
clear s;

% GUARDAR CSV
T = array2table(datos, 'VariableNames', {'Tiempo_s', 'RPM', 'PWM'});
writetable(T, archivoCSV);

disp("✅ Listo. Datos guardados en:");
disp(archivoCSV);
